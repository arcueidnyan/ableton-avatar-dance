cmake_minimum_required(VERSION 3.22)
project(AvatarDance VERSION 1.0.0 LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
include(FetchContent)
set(JUCE_VERSION_TAG 7.0.12)
if(APPLE)
    # Support the current Apple SDK and its ScreenCaptureKit APIs.
    set(JUCE_VERSION_TAG 8.0.8)
endif()
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG ${JUCE_VERSION_TAG} GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(AvatarDance COMPANY_NAME "Avatar Dance" BUNDLE_ID "music.avatardance.vst3"
    PLUGIN_MANUFACTURER_CODE Ardn PLUGIN_CODE Avdc FORMATS VST3
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE PRODUCT_NAME "Avatar Dance")
configure_file(avatars/kira/sprite-sheet.png Kira.png COPYONLY)
configure_file(avatars/blue-static/sprite-sheet.png BlueStatic.png COPYONLY)
juce_add_binary_data(Skins SOURCES "${CMAKE_CURRENT_BINARY_DIR}/Kira.png" "${CMAKE_CURRENT_BINARY_DIR}/BlueStatic.png")
target_sources(AvatarDance PRIVATE vst3/AvatarDance.cpp)
if(APPLE)
    target_sources(AvatarDance PRIVATE vst3/HostVisibility.mm)
endif()
target_compile_features(AvatarDance PRIVATE cxx_std_17)
target_compile_definitions(AvatarDance PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(AvatarDance PRIVATE Skins juce::juce_audio_processors juce::juce_gui_extra)
